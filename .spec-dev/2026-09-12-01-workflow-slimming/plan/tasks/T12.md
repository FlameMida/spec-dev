按 commit-map 回填提交号（文件在镜像仓 `filter-repo/commit-map`：首行 `old new`，其后每行「旧 SHA 新 SHA」；被剪空的提交映射为全零，脚本保留原值并计数）：
```bash
cd "$SRC" && node - /tmp/spec-dev-mirror.git/filter-repo/commit-map <<'JS'
const fs=require("fs"),cp=require("child_process"),ZERO="0".repeat(40);
const map=new Map(fs.readFileSync(process.argv[2],"utf8").trim().split("\n").slice(1).map(l=>l.split(" ")));
const short=new Map([...map].filter(([,n])=>n!==ZERO).map(([o,n])=>[o.slice(0,7),n.slice(0,7)]));
let hits=0,pruned=0;const re40=/\b[0-9a-f]{40}\b/g;
for(const f of cp.execSync("git ls-files .spec-dev",{encoding:"utf8"}).trim().split("\n")){
  const s=fs.readFileSync(f,"utf8");
  const t=s.replace(re40,m=>{const n=map.get(m);if(!n)return m;if(n===ZERO){pruned++;return m;}hits++;return n;});
  if(t!==s)fs.writeFileSync(f,t);}
const c=fs.readFileSync("CHANGELOG.md","utf8");const d=c.replace(/\b[0-9a-f]{7}\b/g,m=>short.has(m)?(hits++,short.get(m)):m);
if(d!==c)fs.writeFileSync("CHANGELOG.md",d);
console.log("remapped",hits,"pruned-refs",pruned);   // 预期 remapped ≥ 200；pruned-refs 逐条人工核对
JS
git ls-files .spec-dev | xargs grep -ohE '\b[0-9a-f]{40}\b' | sort -u | while read h; do git cat-file -e "$h" 2>/dev/null || echo "悬空: $h"; done   # 预期无输出
SKIP_RELEASE_HOOK=1 git commit -am "chore(history): remap commit references after evidence purge" && git push origin main
rm -rf /tmp/spec-dev-mirror.git
```
