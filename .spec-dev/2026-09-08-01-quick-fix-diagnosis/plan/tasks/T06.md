**步骤 3：合并回实际来源**

以下脚本从binding读取来源，验证当前检出/分支/干净状态。来源HEAD不是实施HEAD祖先时，先在实施worktree集成来源变化并复验，不直接合并未经验证的新组合。冲突停止并保存实际阻塞事实，不force或自动解冲突。

```python
import json,subprocess
from pathlib import Path
root=Path.cwd().resolve();F=Path(".spec-dev/2026-09-08-01-quick-fix-diagnosis")
s=json.loads((F/"plan/progress.yaml").read_text())

assert str(root)==b["worktree"]

def git(*args,cwd=root):
    return subprocess.check_output(["rtk","proxy","git",*args],cwd=cwd,text=True).strip()
assert git("branch","--show-current")==b["branch"]
assert git("branch","--show-current",cwd=source)==b["source_branch"]
assert not git("status","--porcelain=v1") and not git("status","--porcelain=v1",cwd=source)
candidate=git("rev-parse","HEAD")
if b["owned"]:
    subprocess.run(["rtk","proxy","git","merge-base","--is-ancestor",git("rev-parse","HEAD",cwd=source),candidate],cwd=root,check=True)
    # If source advanced outside candidate ancestry, integrate it in the owned worktree and revalidate first.
    subprocess.run(["rtk","proxy","env","SKIP_RELEASE_HOOK=1","git","merge","--no-ff",b["branch"],
                    "-m","Merge verified quick-fix diagnosis"],cwd=source,check=True)
    merged=git("rev-parse","HEAD",cwd=source)
else:
    receipt=root/F/"execution/serial/T06/external-merge-receipt.json"
    assert receipt.exists(),"Retained isolation: wait for actual original-mechanism merge receipt; keep T06 in_progress"
    r=json.loads(receipt.read_text())
    assert r["source"]==str(source) and r["source_branch"]==b["source_branch"]
    merged=r["merge_commit"]
    subprocess.run(["rtk","proxy","git","merge-base","--is-ancestor",merged,"HEAD"],cwd=source,check=True)
product=[p for p in git("ls-files").splitlines() if not p.startswith(".spec-dev/")]
assert product
assert not git("diff","--name-only",candidate,merged,"--",*product,cwd=source),"Merged product differs: revalidate before cleanup"

out.parent.mkdir(parents=True,exist_ok=True)
out.write_text(json.dumps({"source":str(source),"source_branch":b["source_branch"],"candidate":candidate,
    "merge_commit":merged,"product_tree_verified":True,"owned_worktree":b["owned"]},ensure_ascii=False,indent=2)+"\n")

print("CONTINUE_FROM_SOURCE",source)
```
