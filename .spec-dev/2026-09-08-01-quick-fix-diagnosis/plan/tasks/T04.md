```python
import json,re
from pathlib import Path
spec=Path(".spec-dev/2026-09-08-01-quick-fix-diagnosis/spec/quick-fix-diagnosis-design.md").read_text()
p=Path("skills/quick-fix/evals/evals.json");data=json.loads(p.read_text())
for sid in ["S24"]:
    m=re.search(r"^#### Scenario: "+sid+r" ([^\n]+)\n(.*?)(?=\n#### Scenario:|\n### Requirement:|\n## |\Z)",spec,re.M|re.S)
    assert m
    parts={k:re.search(r"- \*\*"+k+r"\*\* (.+)",m.group(2)).group(1) for k in ["GIVEN","WHEN","THEN"]}
    name="qf-diagnosis-"+sid.lower()
    assert not any(x["id"]==name for x in data["evals"]),"already materialized; inspect recovery"
    data["evals"].append({"id":name,"prompt":m.group(1)+"\nGIVEN "+parts["GIVEN"]+"\nWHEN "+parts["WHEN"],"expected_output":parts["THEN"]})
p.write_text(json.dumps(data,ensure_ascii=False,indent=2)+"\n")
```
