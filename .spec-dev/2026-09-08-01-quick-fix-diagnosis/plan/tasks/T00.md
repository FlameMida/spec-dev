**步骤 1：检测已有隔离并绑定实际来源**

先在当前仓库根运行以下完整Python块。它读取实际Git路径，非隔离时创建本计划worktree；已隔离仅复用，不跳过绑定或基线。若已隔离且缺原来源记录，先从原机制取证后设置QF_SOURCE_PATH再运行，不能推算main。创建失败记录blocked及实际残留，不force清理；权限拒绝按using-git-worktrees授权纪律处理，不因本机可正常隔离而主动降级。已完成T00恢复直接核对其原binding，不重跑创建块。

```python
import json,os,subprocess
from pathlib import Path
F=Path(".spec-dev/2026-09-08-01-quick-fix-diagnosis")
root=Path.cwd().resolve()
def run(*args,cwd=root):
    return subprocess.check_output(["rtk","proxy","git",*args],cwd=cwd,text=True).strip()
assert run("rev-parse","--show-toplevel")==str(root)
state=json.loads((root/F/"plan/progress.yaml").read_text())
assert state["tasks"]["T00"]["status"] in ["pending","in_progress","blocked"]
gd=run("rev-parse","--path-format=absolute","--git-dir")
common=run("rev-parse","--path-format=absolute","--git-common-dir")
isolated=gd!=common and not run("rev-parse","--show-superproject-working-tree")
old=[json.loads(n[len("binding:"):]) for n in state["notes"] if n.startswith("binding:")]
if isolated:
    if old:
        binding=old[-1]
        assert binding["worktree"]==str(root) and binding["branch"]==run("branch","--show-current")
        wt=root
    else:
        assert os.environ.get("QF_SOURCE_PATH"),"Existing isolation: supply verified source mechanism record in QF_SOURCE_PATH"
        source=Path(os.environ["QF_SOURCE_PATH"]).resolve()
        assert run("rev-parse","--path-format=absolute","--git-common-dir",cwd=source)==common
        binding={"source":str(source),"source_branch":run("branch","--show-current",cwd=source),
                 "base":run("rev-parse","HEAD"),"worktree":str(root),"branch":run("branch","--show-current"),"owned":False}
        assert binding["source_branch"]
        wt=root
else:
    assert not old,"Recorded isolation exists: recover it; do not create another"
    assert not run("status","--porcelain=v1"),"Resolve existing dirty ownership before isolation"
    branch="plan/2026-09-08-01-quick-fix-diagnosis"
    wt=root/".worktrees/plan-2026-09-08-01-quick-fix-diagnosis"
    assert not wt.exists()
    subprocess.run(["rtk","proxy","git","check-ignore","-q",".worktrees/"],cwd=root,check=True)
    source_branch=run("branch","--show-current");assert source_branch
    binding={"source":str(root),"source_branch":source_branch,"base":run("rev-parse","HEAD"),
             "worktree":str(wt),"branch":branch,"owned":True}
    try:subprocess.run(["rtk","proxy","git","worktree","add",str(wt),"-b",branch],cwd=root,check=True)
    except subprocess.CalledProcessError:
        state["tasks"]["T00"]["status"]="blocked";state["current"]=None
        state["notes"].append("T00 isolation creation failed; inspect partial branch/worktree; no cleanup attempted")
        p=root/F/"plan/progress.yaml";tmp=p.with_suffix(".tmp")
        tmp.write_text(json.dumps(state,ensure_ascii=False,indent=2)+"\n");os.replace(tmp,p)
        subprocess.run(["rtk","proxy","git","add",str(F/"plan/progress.yaml")],cwd=root,check=True)
        subprocess.run(["rtk","proxy","env","SKIP_RELEASE_HOOK=1","git","commit","-m","chore(T00): record failed isolation"],cwd=root,check=True)
        raise
state["notes"].append("binding:"+json.dumps(binding,ensure_ascii=False))
state["tasks"]["T00"]["status"]="in_progress";state["current"]="T00"
planned="planned-worktree: .worktrees/plan-2026-09-08-01-quick-fix-diagnosis —— T00核实后替换为实际所有权"
if planned in state["resources"]:state["resources"].remove(planned)
resource=("worktree: "+str(wt)+" —— git worktree remove; git branch -d "+binding["branch"]
          if binding["owned"] else "retained-worktree: "+str(wt)+" —— 交回原隔离机制，不自行删除")
if resource not in state["resources"]:state["resources"].append(resource)
p=wt/F/"plan/progress.yaml";tmp=p.with_suffix(".tmp")
tmp.write_text(json.dumps(state,ensure_ascii=False,indent=2)+"\n");os.replace(tmp,p)
subprocess.run(["rtk","proxy","git","add",str(F/"plan/progress.yaml")],cwd=wt,check=True)
subprocess.run(["rtk","proxy","env","SKIP_RELEASE_HOOK=1","git","commit","-m","chore(T00): bind actual isolated workspace"],cwd=wt,check=True)
print(json.dumps(binding,ensure_ascii=False,indent=2))
```

state.py：
```python
import argparse,json,os,subprocess,time
from pathlib import Path
F=Path(".spec-dev/2026-09-08-01-quick-fix-diagnosis")
p=F/"plan/progress.yaml"
a=argparse.ArgumentParser()
a.add_argument("task",choices=[f"T{i:02}" for i in range(7)])
a.add_argument("status",choices=["pending","in_progress","blocked","completed"])
a.add_argument("--commit");a.add_argument("--tests")
a.add_argument("--note",action="append",default=[])
a.add_argument("--resource",action="append",default=[])
a.add_argument("--drop-resource",action="append",default=[])
v=a.parse_args()
s=json.loads(p.read_text());assert s["format_version"]==1
assert not any(k!=v.task and x["status"]=="in_progress" for k,x in s["tasks"].items())
if v.status=="completed":
    assert v.commit and v.tests=="pass"
    subprocess.run(["rtk","proxy","git","cat-file","-e",v.commit+"^{commit}"],check=True)
    subprocess.run(["rtk","proxy","git","merge-base","--is-ancestor",v.commit,"HEAD"],check=True)
    s["tasks"][v.task].update(commit=v.commit,tests=v.tests)
s["tasks"][v.task]["status"]=v.status
s["current"]=v.task if v.status=="in_progress" else None
s["notes"].extend(v.note)
for r in v.resource:
    if r not in s["resources"]:s["resources"].append(r)
for r in v.drop_resource:
    assert r in s["resources"]
    s["resources"].remove(r)
tmp=p.with_name(p.name+"."+str(os.getpid())+".tmp")
with tmp.open("x") as out:
    out.write(json.dumps(s,ensure_ascii=False,indent=2)+"\n")
    out.flush();os.fsync(out.fileno())
os.replace(tmp,p)
print(json.dumps({"task":v.task,"status":v.status},ensure_ascii=False))
```

record.py：
```python
import argparse,datetime,json,subprocess,time
from pathlib import Path
a=argparse.ArgumentParser()
a.add_argument("task");a.add_argument("label");a.add_argument("command",nargs=argparse.REMAINDER)
v=a.parse_args();assert v.task in [f"T{i:02}" for i in range(7)]
assert v.label.replace("-","").replace("_","").isalnum()
cmd=v.command[1:] if v.command[:1]==["--"] else v.command
assert cmd and cmd[0]=="rtk"
root=Path.cwd().resolve()
out=root/".spec-dev/2026-09-08-01-quick-fix-diagnosis/execution/serial"/v.task/v.label
out.mkdir(parents=True,exist_ok=False)
start=time.monotonic()
with (out/"stdout.log").open("w") as so,(out/"stderr.log").open("w") as se:
    proc=subprocess.Popen(cmd,stdout=so,stderr=se,cwd=root)
    while proc.poll() is None:
        try:proc.wait(timeout=30)
        except subprocess.TimeoutExpired:print(v.task,v.label,"running",round(time.monotonic()-start),flush=True)
facts={"command":cmd,"cwd":str(root),"exit":proc.returncode,"elapsed":time.monotonic()-start,
       "head":subprocess.check_output(["rtk","proxy","git","rev-parse","HEAD"],text=True).strip(),
       "recorded_at":datetime.datetime.now(datetime.timezone.utc).isoformat(),"semantic_verdict":"not_inferred"}
(out/"facts.json").write_text(json.dumps(facts,ensure_ascii=False,indent=2)+"\n")
raise SystemExit(proc.returncode)
```


状态工具是本计划的串行记录代码，不是quick-fix产品接口。无需为治理写入制造TDD红。
