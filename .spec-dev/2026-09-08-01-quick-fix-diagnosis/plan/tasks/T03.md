    parts={k:re.search(r"- \*\*"+k+r"\*\* (.+)",m.group(2)).group(1) for k in ["GIVEN","WHEN","THEN"]}

p.write_text(json.dumps(data,ensure_ascii=False,indent=2)+"\n")

import json
from pathlib import Path
edits=[
  {

    "old": "### 步骤 6：可选自动化验收",
