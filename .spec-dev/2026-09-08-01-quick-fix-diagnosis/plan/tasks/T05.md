**步骤 3：检查原件完整性与资源归属**

先对每个process.json使用其实际PID/PGID、启动/会话信息核查进程，命令为rtk proxy ps -p PID -o pid=,pgid=,lstart=,command=；PID不存在可记录退出，PID复用不能误杀，身份未核实不自动终止。仍有本次writer则等待/按其实际异常处理，不进行快照清理或全量。

下面代码验证每个登记fixture的活文件与已保存快照一致、Git bundle可用，生成供T06消费的具体清理凭据，不删除资源：
```python
import hashlib,json,tarfile,subprocess,uuid
from pathlib import Path
root=Path.cwd().resolve()
F=root/".spec-dev/2026-09-08-01-quick-fix-diagnosis"
state=json.loads((F/"plan/progress.yaml").read_text())
records=[]
for resource in state["resources"]:
    if not resource.startswith("fixture: "):continue
    wt=Path(resource.split(" —— ",1)[0][len("fixture: "):]).resolve()
    assert wt.is_relative_to(root/".worktrees/qf-qa") and wt.is_dir() and not wt.is_symlink()
    assert subprocess.check_output(["rtk","proxy","git","rev-parse","--show-toplevel"],cwd=wt,text=True).strip()==str(wt)
    live={str(p.relative_to(wt)):hashlib.sha256(p.read_bytes()).hexdigest()
          for p in wt.rglob("*") if p.is_file() and ".git" not in p.parts}
    matches=[]
    for f in (F/"acceptance/model").rglob("facts.json"):
        data=json.loads(f.read_text())
        if data.get("fixture")!=str(wt):continue
        archive=f.parent/"fixture.tar.gz";bundle=f.parent/"fixture.bundle"
        if not archive.exists() or not bundle.exists():continue
        with tarfile.open(archive) as tar:
            members=tar.getmembers()
            assert all(m.isfile() and ".." not in Path(m.name).parts and not Path(m.name).is_absolute() for m in members)
            archived={m.name:hashlib.sha256(tar.extractfile(m).read()).hexdigest() for m in members}
        if archived==live:matches.append((f.stat().st_mtime,archive,bundle))
    if not matches:
        # Stopped/failed runner may leave a real owned fixture without its final snapshot.
        # Preserve the current bytes as recovery evidence; this is never a model PASS.
        d=F/"acceptance/model/retained-archives"/str(uuid.uuid4());d.mkdir(parents=True)
        archive=d/"fixture.tar.gz";bundle=d/"fixture.bundle"
        assert not any(x.is_symlink() for x in wt.rglob("*"))
        with tarfile.open(archive,"w:gz") as tar:
            for rel in sorted(live):tar.add(wt/rel,arcname=rel,recursive=False)
        subprocess.run(["rtk","proxy","git","bundle","create",str(bundle),"--all"],cwd=wt,check=True)
        (d/"facts.json").write_text(json.dumps({"fixture":str(wt),"mode":"host_recovery_archive","verdict":"unverified","reason":"Stopped runner or changed live fixture; preserve bytes, not new model evidence"},indent=2)+"\n")
        matches.append((0,archive,bundle))
    _,archive,bundle=max(matches)
    subprocess.run(["rtk","proxy","git","bundle","verify",str(bundle)],cwd=wt,check=True)
    records.append({"resource":resource,"path":str(wt),"archive":str(archive.relative_to(F)),
        "archive_sha256":hashlib.sha256(archive.read_bytes()).hexdigest(),
        "bundle":str(bundle.relative_to(F)),"bundle_sha256":hashlib.sha256(bundle.read_bytes()).hexdigest(),
        "files":live,"head":subprocess.check_output(["rtk","proxy","git","rev-parse","HEAD"],cwd=wt,text=True).strip()})
p=F/"acceptance/resources.json";assert not p.exists()
p.write_text(json.dumps(records,ensure_ascii=False,indent=2)+"\n")
print("Verified snapshot and bundle for",len(records),"owned fixture roots; no cleanup executed")
```

独立审查者再核对资源与本任务创建记录、原始输出、快照一致；报告中说明用户工作/无归属目录保留。本特性snapshot包含原始失败和未提交修复，不因选择成功证据丢掉它们。中断后新增的host_recovery_archive只证明归档当前字节，不能恢复或提升失败run的模型结论；Git初始化不完整或归属不明仍保持阻塞并保留现场。
