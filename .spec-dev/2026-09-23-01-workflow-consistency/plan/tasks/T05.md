## 步骤 1：安装后真实 Git 行为测试

在 guard-install.test.mjs 使用 workflowFixture；在测试目录先 unset core.hooksPath（夹具初始 /dev/null），随后运行实际 install.mjs，保留一个自定义 hook 哨兵。
```js
test('S06 installed guard survives without plugin cache and stamps task commit',t=>{
 const f=workflowFixture(t);f.git('config','--unset','core.hooksPath');
 const install=f.run('guardrail/install.mjs',['--repo',f.root,'--no-ci','--no-migrate']);
 assert.equal(install.status,0,install.stderr);
 const binding=path.join(f.root,'scripts/spec-dev/task-binding.mjs');
 assert.ok(existsSync(binding),'installed binding CLI is required');
 // 用已建立的真实 binding authority 调安装后的 bind；关联来源不由测试伪造通过。
 const authority=f.bind().authority;
 const b=spawnSync(process.execPath,[binding,'bind','--plan',f.plan,'--task','T01','--authority',authority],{cwd:f.root,encoding:'utf8'});
 assert.equal(b.status,0,b.stderr);
 f.put('src/app.mjs','export const value=3;\n');f.git('add','src/app.mjs');f.git('commit','-qm','implementation');
 assert.match(f.git('log','-1','--format=%B'),/^Spec-Task: /m);
});
```
追加已有相同标记不重复、冲突/多个标记拒绝、普通 Spec 无绑定不自动升级、二次安装幂等、自定义 hook 哨兵仍执行。所有新 remote/clone 只用本票拥有的本地 bare 仓库，不接真实远端。

## 步骤 3：安装与消息协议

- 安装器只复制 guardrail 的核心、task-binding 及它们直接使用的 lib 文件；不复制 validate-output、全部 schemas 或 scripts/lib。
- 新消息 hook 通过安装路径调用 binding CLI。prepare 只补缺失 Spec-Task；check 再验证 authority 的 task/plan 与当前有效回执，拒绝重复、JSON 非法及不匹配。
- 不能在 pre-commit 猜未来 commit message；pre-commit 检查 index 范围，commit-msg 检查持久关联。hook 出错退出非零，不能吃掉确定性绑定拒绝。
- 使用 installGitHooks 既有标记区块机制合并两个新 hook；保留用户原内容、可执行权限和返回码，重装不堆叠。
- 消息文件先规范化并核对属于本工作区真实 Git dir 的消息文件，拒绝路径逃逸/符号链接到外部；该 CLI 不成为任意文件追加工具。

消息注入主体：
```js
export function addTaskTrailer(message,reference){
 const lines=message.split(/\r?\n/).filter(l=>/^Spec-Task:/i.test(l));
 if(lines.length>1)throw new Error('duplicate Spec-Task');
 const expected={plan:reference.plan,task:reference.task,authority:reference.authority};
 if(lines.length){
  const actual=parseUniqueJson(lines[0].slice(lines[0].indexOf(':')+1).trim());
  if(Object.keys(actual).length!==3||Object.entries(expected).some(([key,value])=>actual[key]!==value))throw new Error('Spec-Task mismatch');
  return message;
 }
 return message.trimEnd()+'\n\nSpec-Task: '+JSON.stringify(expected)+'\n';
}
```
字段比较实现时按固定三个键比较，不将纯 JSON 键顺序差异当权限差异；拒绝额外键。CI 模板保留所需 Git 历史；缺源对象明确失败，不联网猜未知来源或自动发布本地归档 ref。最终 squash 的来源映射由 T09/T14 验证。
