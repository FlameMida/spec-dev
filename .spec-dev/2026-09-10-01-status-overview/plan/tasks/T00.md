```python
             "worktree":str(wt),"branch":branch,"owned":True}

        state["notes"].append("T00 isolation creation failed; inspect partial branch/worktree; no cleanup attempted")

resource=("worktree: "+str(wt)+" —— git worktree remove; git branch -d "+binding["branch"]
          if binding["owned"] else "retained-worktree: "+str(wt)+" —— 交回原隔离机制，不自行删除")
if resource not in state["resources"]:state["resources"].append(resource)
planned_fixtures="planned-fixtures: .worktrees/status-overview-qa —— 位于实际实施工作区内，测试/模型目录创建前登记，T07仅清理本计划所属资源"
```

```python
import argparse,json,os,subprocess,time
from pathlib import Path
F=Path(".spec-dev/2026-09-10-01-status-overview")
p=F/"plan/progress.yaml"
a=argparse.ArgumentParser()
a.add_argument("task",choices=[f"T{i:02}" for i in range(8)])
a.add_argument("status",choices=["pending","in_progress","blocked","completed"])
a.add_argument("--commit");a.add_argument("--tests")
a.add_argument("--note",action="append",default=[])
a.add_argument("--resource",action="append",default=[])
a.add_argument("--drop-resource",action="append",default=[])
v=a.parse_args()
s=json.loads(p.read_text());assert s["format_version"]==1
assert not any(k!=v.task and x["status"]=="in_progress" for k,x in s["tasks"].items())
if v.status=="completed":
    assert v.commit and v.tests=="pass"
    assert all(s["tasks"][f"T{i:02}"]["status"]=="completed" for i in range(int(v.task[1:])))
    subprocess.run(["rtk","proxy","git","cat-file","-e",v.commit+"^{commit}"],check=True)
    subprocess.run(["rtk","proxy","git","merge-base","--is-ancestor",v.commit,"HEAD"],check=True)
    s["tasks"][v.task].update(commit=v.commit,tests=v.tests)
s["tasks"][v.task]["status"]=v.status
s["current"]=v.task if v.status=="in_progress" else None
s["notes"].extend(v.note)
for r in v.resource:
    if r not in s["resources"]:s["resources"].append(r)
for r in v.drop_resource:
    assert r in s["resources"]
    s["resources"].remove(r)
tmp=p.with_name(p.name+"."+str(os.getpid())+".tmp")
with tmp.open("x") as out:
    out.write(json.dumps(s,ensure_ascii=False,indent=2)+"\n")
    out.flush();os.fsync(out.fileno())
os.replace(tmp,p)
print(json.dumps({"task":v.task,"status":v.status},ensure_ascii=False))
```

```python
out.mkdir(parents=True,exist_ok=False)
start=time.monotonic()
with (out/"stdout.log").open("w") as so,(out/"stderr.log").open("w") as se:
    proc=subprocess.Popen(cmd,stdout=so,stderr=se,cwd=root)
    while proc.poll() is None:
        try:proc.wait(timeout=30)
        except subprocess.TimeoutExpired:print(v.task,v.label,"running",round(time.monotonic()-start),flush=True)
facts={"command":cmd,"cwd":str(root),"exit":proc.returncode,"elapsed":time.monotonic()-start,
       "head":subprocess.check_output(["rtk","proxy","git","rev-parse","HEAD"],text=True).strip(),
       "recorded_at":datetime.datetime.now(datetime.timezone.utc).isoformat(),"semantic_verdict":"not_inferred"}
(out/"facts.json").write_text(json.dumps(facts,ensure_ascii=False,indent=2)+"\n")
raise SystemExit(proc.returncode)
```
